  var ARTEN={'t-shirt':'T-Shirt','fotodruck':'Fotodruck','stofftasche':'Stofftasche','sticker':'Sticker','holografischer-aufkleber':'Holo-Sticker','dekokissen':'Kissen','grußkarte':'Grußkarte','notizbuch':'Notizbuch','cap':'Cap','pullover':'Pullover','poster':'Poster','tasse':'Tasse','iphone-hülle':'iPhone-Hülle','acrylblock':'Acrylblock','täschchen':'Täschchen','fleecedecke':'Decke','mauspad':'Mauspad','wandtuch':'Wandtuch','hoodie':'Hoodie','leinwandbild':'Leinwand','samsung-hülle':'Samsung-Hülle','kinder-t-shirt':'Kinder-Shirt','button':'Button','kunstdruck':'Kunstdruck','magnet':'Magnet','langarmshirt':'Langarmshirt','postkarte':'Postkarte','gerahmtes-bild':'Gerahmtes Bild','untersetzer':'Untersetzer','socken':'Socken','tank-top':'Tank Top','kleid':'Kleid','baby-body':'Baby-Body'};

  function bild(s){return s.charAt(0)==='/'?s:IMG+s}
  function esc(t){return String(t).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/"/g,'&quot;')}
  function art(slug){var s=decodeURIComponent(slug);return ARTEN[s]||s.charAt(0).toUpperCase()+s.slice(1)}

  var felder={};
  function feld(key){
    var w=D[key], m=META[key]; if(!w||!m) return;
    var n=m.th?6:7;
    var karten=w.p.slice(0,n).map(function(p){
      var h=p[2]?RB+p[2]:COL+w.col;
      return '<a class="xk" href="'+esc(h)+'" target="_blank" rel="noopener"><img src="'+esc(bild(p[3]))+'" alt="'+esc(p[0]+', '+p[1])+'" loading="lazy"><span class="xn">'+esc(p[0])+'</span><span class="xp">'+esc(p[1])+'</span></a>';
    }).join('');
    var thema=m.th?'<a class="wthema" href="'+m.th[0]+'"><img src="'+esc(m.th[2])+'" alt="" loading="lazy"><span><span class="k">Themenseite</span><b>Zur Themenseite '+esc(m.th[1])+' →</b></span></a>':'';
    var sec=document.createElement('section');
    sec.className='wfeld'; sec.id=m.id||('w-'+key); sec.style.background=m.f;
    sec.innerHTML='<div class="xkopf"><div><span class="k">Welt · Redbubble</span><h2 class="h2">'+esc(w.t)+' <span lang="ko" style="font-weight:700">'+m.ko+'</span></h2></div>'
      +'<div class="xr"><p>'+esc(m.x)+'</p><a class="knopf" href="'+COL+w.col+'" target="_blank" rel="noopener">Ganze Kollektion →</a></div></div>'
      +'<div class="wreihe'+(m.th?'':' ohne')+'">'+karten+thema+'</div>'
      +'<details class="alle"><summary>Alle Redbubble-Designs aus '+esc(w.t)+' ansehen</summary><div class="raster"></div><button class="mehrladen" type="button" hidden>Mehr anzeigen</button></details>';
    ziel.appendChild(sec);
    var f={key:key,el:sec,det:sec.querySelector('details'),sum:sec.querySelector('summary'),ras:sec.querySelector('.raster'),btn:sec.querySelector('.mehrladen'),liste:[],gezeigt:0};
    f.det.addEventListener('toggle',function(){if(f.det.open&&!f.gezeigt) mehr(f)});
    f.btn.addEventListener('click',function(){mehr(f)});
    felder[key]=f;
  }

  var SCHRITT=24;
  function mehr(f){
    f.liste=daten().W[f.key]||[];
    var teil=f.liste.slice(f.gezeigt,f.gezeigt+SCHRITT);
    f.ras.insertAdjacentHTML('beforeend',teil.map(function(z){
      var g=z.split('|'), t=daten().T[+g[1]], b=g[4].split('/'), kopf=b[0], rest=b.slice(1).join('/');
      if(/^\d+$/.test(rest)) rest=daten().S[+rest];
      var src=(kopf.indexOf('0:')===0?'https://ih0.redbubble.net/image.'+kopf.slice(2):IMG+kopf)+'/'+rest;
      return '<a class="pk" href="'+RB+t+'/Jasardpu/'+g[2]+'/'+g[3]+'" target="_blank" rel="noopener"><img src="'+esc(src)+'" alt="'+esc(g[0]+', '+art(t))+'" loading="lazy"><b>'+esc(g[0])+'</b><small>'+esc(art(t))+'</small></a>';
    }).join(''));
    f.gezeigt+=teil.length;
    f.btn.hidden=f.gezeigt>=f.liste.length;
    if(!f.btn.hidden) f.btn.textContent='Mehr anzeigen ('+(f.liste.length-f.gezeigt)+' weitere)';
  }

  function zahlen(){
    FOLGE.forEach(function(k){
      var f=felder[k]; if(!f) return;
      var l=daten().W[k]||[];
      f.det.hidden=!l.length; f.det.style.display=l.length?'':'none';
      if(l.length) f.sum.textContent='Alle '+l.length+' Redbubble-Designs aus '+D[k].t+' ansehen';
    });
  }

  FOLGE.forEach(feld);
  zahlen();
  window.jasardpuDatenDa=zahlen;
})();
